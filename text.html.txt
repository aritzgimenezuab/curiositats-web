        // Funció per carregar la graella d'articles al llistat principal
        function carregarArticles(filtre = 'totes') {
            const graella = document.getElementById('graella-articles');
            graella.innerHTML = '';

            const articlesFiltrats = filtre === 'totes' 
                ? articles 
                : articles.filter(a => a.categoria === filtre);

            articlesFiltrats.forEach(article => {
                let colorCategoria = "bg-indigo-50 text-indigo-600";
                if(article.categoria === "Història") colorCategoria = "bg-emerald-50 text-emerald-600";
                if(article.categoria === "Ciència") colorCategoria = "bg-amber-50 text-amber-600";

                const targeta = document.createElement('div');
                targeta.className = "bg-white rounded-2xl border border-slate-200 p-6 shadow-sm hover:shadow-md transition cursor-pointer flex flex-col justify-between";
                targeta.onclick = () => obrirArticle(article.id);
                
                targeta.innerHTML = `
                    <div>
                        <div class="flex items-center space-x-2 text-xs font-semibold mb-3">
                            <span class="${colorCategoria} px-2.5 py-1 rounded-full">${article.categoria}</span>
                            <span class="text-slate-400">•</span>
                            <span class="text-slate-500">${article.temps}</span>
                        </div>
                        <h3 class="text-xl font-bold text-slate-900 mb-2 hover:text-indigo-600 transition">
                            ${article.titol}
                        </h3>
                        <p class="text-slate-600 text-sm leading-relaxed mb-4">
                            ${article.resum}
                        </p>
                    </div>
                    <div class="flex items-center text-indigo-600 font-semibold text-sm">
                        Llegir article complet &rarr;
                    </div>
                `;
                graella.appendChild(targeta);
            });
        }

        // Funció per obrir el detall d'un article en concret
        function obrirArticle(id) {
            const article = articles.find(a => a.id === id);
            if (!article) return;

            document.getElementById('vista-llistat').classList.add('hidden');
            document.getElementById('vista-detall').classList.remove('hidden');
            document.getElementById('detall-contingut').innerHTML = article.contingut;
            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        // Funció per tornar a la vista d'inici / llistat
        function mostrarInici(event) {
            if (event) event.preventDefault();
            document.getElementById('vista-detall').classList.add('hidden');
            document.getElementById('vista-llistat').classList.remove('hidden');
            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        // Funció per filtrar per categoria des del menú superior
        function filtrarCategoria(categoria, event) {
            event.preventDefault();
            mostrarInici();
            carregarArticles(categoria);
            
            const textFiltre = document.getElementById('filtre-actual');
            textFiltre.innerText = categoria === 'totes' ? 'Mostrant tot el contingut' : `Categoria: ${categoria}`;
        }

        // Inicialitzar la pàgina carregant els articles en entrar
        window.onload = () => {
            carregarArticles('totes');
